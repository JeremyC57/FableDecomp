# Linux / Android build of the recompiled game: the shared host (../win) on the POSIX
# Win32 layer (w32/), with DXVK Native for Direct3D 9 and SDL2 for windows, input and audio.
#
#   cmake -S rebuild/recomp/posix -B build/linux -G Ninja -DCMAKE_BUILD_TYPE=Release \
#         -DFABLE_GEN_DIR=<lifter output> "-DFABLE_GUEST_DLLS=cfgdetect|ConfigDetect.dll|<dir>" \
#         -DMINGW_HEADERS=<mingw-w64>/include -DDXVK_D3D9=<dxvk build>/src/d3d9/libdxvk_d3d9.so
#   cmake --build build/linux
cmake_minimum_required(VERSION 3.22)
project(FableRecompPosix LANGUAGES C CXX ASM)

set(FABLE_GEN_DIR "" CACHE PATH "Directory with recomp_*.c generated by fable_recomp")
set(MINGW_HEADERS "" CACHE PATH "MinGW-w64 include directory (Windows API declarations)")
set(DXVK_D3D9 "" CACHE FILEPATH "DXVK Native libdxvk_d3d9.so")
if(NOT EXISTS "${FABLE_GEN_DIR}/recomp_table.c")
  message(FATAL_ERROR "FABLE_GEN_DIR must point at the lifter output (recomp_table.c not found)")
endif()
if(NOT EXISTS "${MINGW_HEADERS}/windows.h")
  message(FATAL_ERROR "MINGW_HEADERS must point at a MinGW-w64 include directory")
endif()

set(WIN_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../win)
set(RUNTIME_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../runtime)

# ---- generated code (same as the Windows build) ------------------------------------------
file(GLOB GEN_SOURCES "${FABLE_GEN_DIR}/recomp_*.c")
add_library(fable_gen OBJECT ${GEN_SOURCES} ${RUNTIME_DIR}/runtime.c)
target_include_directories(fable_gen PRIVATE ${RUNTIME_DIR} ${FABLE_GEN_DIR})
target_compile_definitions(fable_gen PRIVATE RECOMP_IDENTITY_MEMORY=1)
set(FABLE_GEN_OPT "-O1" CACHE STRING "Optimisation level for the lifted code (-O2 for the handheld builds)")
separate_arguments(FABLE_GEN_OPT_LIST UNIX_COMMAND "${FABLE_GEN_OPT}")
target_compile_options(fable_gen PRIVATE ${FABLE_GEN_OPT_LIST} -fsigned-char -ffp-contract=off -fno-strict-aliasing -w)
# ThinLTO across the lifted code: direct calls between small lifted functions inline, which
# keeps the guest registers in host registers across them instead of spilling to the Ctx.
option(FABLE_GEN_LTO "ThinLTO for the lifted code (needs lld)" OFF)
if(FABLE_GEN_LTO)
  target_compile_options(fable_gen PRIVATE -flto=thin)
endif()

set(FABLE_GUEST_DLLS "" CACHE STRING "prefix|file|dir;... for recompiled DLLs")
set(GUEST_DLL_DECLS "")
set(GUEST_DLL_ROWS "")
set(GUEST_DLL_COUNT 0)
foreach(spec ${FABLE_GUEST_DLLS})
  string(REPLACE "|" ";" parts "${spec}")
  list(GET parts 0 pfx)
  list(GET parts 1 dllfile)
  list(GET parts 2 dlldir)
  file(GLOB dll_sources "${dlldir}/${pfx}_*.c")
  target_sources(fable_gen PRIVATE ${dll_sources})
  target_include_directories(fable_gen PRIVATE ${dlldir})
  string(APPEND GUEST_DLL_DECLS "extern const RecompEntry recomp_table_${pfx}[]; extern const uint32_t recomp_table_${pfx}_size;\n")
  string(APPEND GUEST_DLL_ROWS "    {\"${dllfile}\", recomp_table_${pfx}, &recomp_table_${pfx}_size},\n")
  math(EXPR GUEST_DLL_COUNT "${GUEST_DLL_COUNT} + 1")
endforeach()
if(GUEST_DLL_COUNT EQUAL 0)
  set(GUEST_DLL_ROWS "    {0, 0, 0},\n")
endif()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/guest_dlls.c
  "/* Generated by CMake: recompiled DLL tables. */\n#include <stdint.h>\n"
  "typedef void (*GuestFn)(void*);\ntypedef struct RecompEntry { uint32_t addr; GuestFn fn; } RecompEntry;\n"
  "${GUEST_DLL_DECLS}"
  "typedef struct GuestDllTable { const char* file; const RecompEntry* table; const uint32_t* size; } GuestDllTable;\n"
  "const GuestDllTable guest_dll_tables[] = {\n${GUEST_DLL_ROWS}};\nconst uint32_t guest_dll_count = ${GUEST_DLL_COUNT};\n")
target_sources(fable_gen PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/guest_dlls.c)

# ---- host -----------------------------------------------------------------------------------
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)

file(GLOB HOST_SOURCES ${WIN_DIR}/*.cpp)
file(GLOB W32_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/w32/*.cpp ${CMAKE_CURRENT_SOURCE_DIR}/w32/*.c ${CMAKE_CURRENT_SOURCE_DIR}/w32/*.S)
file(GLOB GLES_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/gles/*.cpp ${CMAKE_CURRENT_SOURCE_DIR}/overlay/*.cpp)  # Direct3D 9 census, the OpenGL ES renderer, the host overlay menu
set(HOST_ALL ${HOST_SOURCES} ${W32_SOURCES} ${GLES_SOURCES} ${CMAKE_CURRENT_SOURCE_DIR}/main_posix.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/movie_ffmpeg.cpp $<TARGET_OBJECTS:fable_gen>)
if(ANDROID)
  # SDLActivity loads libmain.so and calls SDL_main on its own thread.
  set(ADRENOTOOLS_DIR "" CACHE PATH "libadrenotools checkout (custom Vulkan drivers)")
  set(ADRENOTOOLS_BUILD "" CACHE PATH "libadrenotools build directory")
  add_library(FableRecomp SHARED ${HOST_ALL} ${CMAKE_CURRENT_SOURCE_DIR}/android/android_host.cpp)
  set_target_properties(FableRecomp PROPERTIES OUTPUT_NAME main)
  target_include_directories(FableRecomp PRIVATE ${ADRENOTOOLS_DIR}/include)
  target_link_libraries(FableRecomp PRIVATE ${ADRENOTOOLS_BUILD}/libadrenotools.a
    ${ADRENOTOOLS_BUILD}/lib/linkernsbypass/liblinkernsbypass.a android log)
  target_link_options(FableRecomp PRIVATE -Wl,-z,max-page-size=16384 -static-libstdc++)
else()
  add_executable(FableRecomp ${HOST_ALL})
endif()
target_include_directories(FableRecomp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include ${RUNTIME_DIR} ${WIN_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/gles)
# Windows API declarations come after the system headers, so the C library stays the system one.
target_compile_options(FableRecomp PRIVATE $<$<COMPILE_LANGUAGE:C,CXX>:-idirafter ${MINGW_HEADERS}>)
target_compile_definitions(FableRecomp PRIVATE FABLE_POSIX=1 RECOMP_IDENTITY_MEMORY=1)
target_compile_features(FableRecomp PRIVATE cxx_std_20)
target_compile_options(FableRecomp PRIVATE
  $<$<COMPILE_LANGUAGE:C,CXX>:-fshort-wchar -fdeclspec -ffp-contract=off -fno-strict-aliasing>
  $<$<COMPILE_LANGUAGE:C,CXX>:-Wall -Wno-unused-function -Wno-ignored-attributes -Wno-unknown-pragmas -Wno-builtin-requires-header>
  $<$<COMPILE_LANGUAGE:C,CXX>:-Wno-nonportable-include-path -Wno-pragma-pack -Wno-microsoft-anon-tag -Wno-int-to-pointer-cast -Wno-microsoft-exception-spec>)
target_link_libraries(FableRecomp PRIVATE PkgConfig::SDL2 PkgConfig::FFMPEG dl)
if(FABLE_GEN_LTO)
  target_link_options(FableRecomp PRIVATE -flto=thin -fuse-ld=lld -Wl,--thinlto-jobs=4)
endif()
# ---- code layout (rg34xx/linux-a64.cmake turns on all but the PGO options for the handheld) -
# The lifted code is ~56 MB of .text in guest-address order, so a frame's functions are spread
# over thousands of pages: an in-order Cortex-A53 with 10 L1 iTLB entries and a small L2 pays a
# TLB walk and a cache miss on most calls. These pack the hot functions together (and onto huge
# pages) and let a profile steer the compiler.
option(FABLE_LLD "Link with lld" OFF)
set(FABLE_SYMBOL_ORDER "" CACHE FILEPATH "Hot functions placed first (lld --symbol-ordering-file; rg34xx/gen_hot_order.py writes it)")
option(FABLE_HUGE_TEXT "At startup, move the hot code onto transparent huge pages (codelayout.cpp)" OFF)
set(FABLE_PGO "" CACHE STRING "Clang PGO: 'gen' builds an instrumented binary (training run), or the path of a merged .profdata to optimise with")
set(FABLE_PGO_RUNTIME "" CACHE FILEPATH "libclang_rt.profile archive for FABLE_PGO=gen when clang's own is not installed (libclang-rt-18-dev)")
option(FABLE_A53_ERRATA "Cortex-A53 errata 835769 (compiler) and 843419 (linker) workarounds; GCC applies both by default, clang neither" OFF)
if(NOT ANDROID)  # the hot-range markers, the huge-page remap, FABLE_PMU counters
  target_sources(FableRecomp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/codelayout.cpp)
  set_property(SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/main_posix.cpp APPEND PROPERTY COMPILE_DEFINITIONS FABLE_CODELAYOUT=1)
endif()
if(FABLE_LLD OR FABLE_SYMBOL_ORDER)
  find_program(FABLE_LD_LLD NAMES ld.lld DOC "lld, the linker FABLE_LLD and FABLE_SYMBOL_ORDER use")
  if(FABLE_LD_LLD)
    target_link_options(FableRecomp PRIVATE -fuse-ld=lld)
  else()
    message(WARNING "ld.lld not found (Debian/Ubuntu: apt install lld): linking with the default linker, "
      "without the hot-function order (FABLE_SYMBOL_ORDER) the huge-page remap relies on")
    set(FABLE_SYMBOL_ORDER "")  # GNU ld has no --symbol-ordering-file
  endif()
endif()
if(FABLE_SYMBOL_ORDER)
  if(NOT EXISTS "${FABLE_SYMBOL_ORDER}")
    message(FATAL_ERROR "FABLE_SYMBOL_ORDER: ${FABLE_SYMBOL_ORDER} not found")
  endif()
  # one section per function, so the linker can move each; --no-rosegment keeps the read-only
  # data in the code segment ahead of .text (as GNU ld lays it out), which lets the huge-page
  # range start below the first hot function
  target_compile_options(fable_gen PRIVATE -ffunction-sections)
  target_compile_options(FableRecomp PRIVATE $<$<COMPILE_LANGUAGE:C,CXX>:-ffunction-sections>)
  target_link_options(FableRecomp PRIVATE "LINKER:--symbol-ordering-file=${FABLE_SYMBOL_ORDER}"
    LINKER:--no-warn-symbol-ordering LINKER:--no-rosegment)
  set_property(TARGET FableRecomp APPEND PROPERTY LINK_DEPENDS "${FABLE_SYMBOL_ORDER}")
endif()
if(FABLE_A53_ERRATA)
  target_compile_options(fable_gen PRIVATE -mfix-cortex-a53-835769)
  target_compile_options(FableRecomp PRIVATE $<$<COMPILE_LANGUAGE:C,CXX>:-mfix-cortex-a53-835769>)
  # The A53 has no LSE atomics: inline load/store-exclusive loops instead of clang's outlined
  # __aarch64_* helpers (a call and a runtime check per atomic, e.g. the GL queue's producer lock).
  target_compile_options(fable_gen PRIVATE -mno-outline-atomics)
  target_compile_options(FableRecomp PRIVATE $<$<COMPILE_LANGUAGE:C,CXX>:-mno-outline-atomics>)
  target_link_options(FableRecomp PRIVATE LINKER:--fix-cortex-a53-843419)
endif()
if(FABLE_HUGE_TEXT AND NOT ANDROID)
  set_property(SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/codelayout.cpp APPEND PROPERTY COMPILE_DEFINITIONS FABLE_HUGE_TEXT=1)
endif()
if(FABLE_PGO STREQUAL "gen")
  # the x86-64 harness build: function and branch counts of a play session (rg34xx/pgo_train.sh)
  if(ANDROID)
    message(FATAL_ERROR "FABLE_PGO=gen: the instrumented training build is the Linux (harness) one; "
      "Android builds take a profile from it with FABLE_PGO=<path>.profdata")
  endif()
  target_compile_options(fable_gen PRIVATE -fprofile-instr-generate)
  target_compile_options(FableRecomp PRIVATE $<$<COMPILE_LANGUAGE:C,CXX>:-fprofile-instr-generate>)
  if(FABLE_PGO_RUNTIME)
    target_link_options(FableRecomp PRIVATE -u __llvm_profile_runtime)
    target_link_libraries(FableRecomp PRIVATE ${FABLE_PGO_RUNTIME})
  else()
    target_link_options(FableRecomp PRIVATE -fprofile-instr-generate)
  endif()
  set_property(SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/codelayout.cpp APPEND PROPERTY COMPILE_DEFINITIONS FABLE_PGO_GEN=1)
elseif(FABLE_PGO)
  if(NOT EXISTS "${FABLE_PGO}")
    message(FATAL_ERROR "FABLE_PGO: ${FABLE_PGO} not found")
  endif()
  # the profile comes from the same sources built for another host: functions the lifter has
  # changed since (or that differ per architecture) are simply left unprofiled
  set(PGO_USE -fprofile-instr-use=${FABLE_PGO} -Wno-profile-instr-out-of-date -Wno-profile-instr-unprofiled -Wno-profile-instr-missing)
  target_compile_options(fable_gen PRIVATE ${PGO_USE})
  target_compile_options(FableRecomp PRIVATE "$<$<COMPILE_LANGUAGE:C,CXX>:${PGO_USE}>")
  set_property(SOURCE ${GEN_SOURCES} ${HOST_SOURCES} ${W32_SOURCES} ${GLES_SOURCES}
    APPEND PROPERTY OBJECT_DEPENDS "${FABLE_PGO}")  # a new profile recompiles
endif()

# d3d9.cpp opens DXVK at run time. Linux also links it, so the build directory's copy is found;
# the APK carries two DXVK builds and android_host.cpp picks one.
if(NOT ANDROID)
  target_link_libraries(FableRecomp PRIVATE ${DXVK_D3D9})
endif()
if(NOT ANDROID)
  target_link_libraries(FableRecomp PRIVATE pthread)
endif()
# MSVC's char is signed; so is the guest's.
target_compile_options(FableRecomp PRIVATE $<$<COMPILE_LANGUAGE:C,CXX>:-fsigned-char>)
set_target_properties(FableRecomp PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
