# Recompiled original-Xbox Fable: The Lost Chapters (default.xbe) for Linux and Android.
#
#   fable_recomp <iso>/default.xbe - <gen>
#   cmake -S rebuild/recomp/xbox -B build/xbox -G Ninja -DCMAKE_BUILD_TYPE=Release -DFABLE_GEN_DIR=<gen>
#   cmake --build build/xbox
#   build/xbox/FableXbox --game <iso>
cmake_minimum_required(VERSION 3.22)
project(FableXbox LANGUAGES C CXX)

set(FABLE_GEN_DIR "" CACHE PATH "Directory with recomp_*.c generated by fable_recomp from default.xbe")
if(NOT EXISTS "${FABLE_GEN_DIR}/recomp_table.c")
  message(FATAL_ERROR "FABLE_GEN_DIR must point at the lifter output (recomp_table.c not found)")
endif()
set(RUNTIME_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../runtime)

file(GLOB GEN_SOURCES CONFIGURE_DEPENDS "${FABLE_GEN_DIR}/recomp_*.c")
add_library(fable_xbox_gen STATIC ${GEN_SOURCES} ${RUNTIME_DIR}/runtime.c)
target_include_directories(fable_xbox_gen PRIVATE ${RUNTIME_DIR} ${FABLE_GEN_DIR})
# Optimisation level of the lifted game code (most of the CPU time; the movie decoder's MMX/SSE
# colour conversion especially benefits from vectorisation at -O2).
set(FABLE_GEN_OPT "-O1" CACHE STRING "Optimisation flag for the lifted code")
target_compile_options(fable_xbox_gen PRIVATE ${FABLE_GEN_OPT} -fsigned-char -ffp-contract=off -fno-strict-aliasing -w)

set(HOST_SOURCES core.cpp memory.cpp kernel.cpp files.cpp disc.cpp profiler.cpp xc.cpp main.cpp hle.cpp xmv.cpp audio.cpp coro.cpp settings.cpp
    nv2a.cpp soft_renderer.cpp video.cpp vk.cpp input.cpp
    vk_renderer.cpp vk_pipeline.cpp nv2a_vsh.cpp nv2a_psh.cpp nv2a_texture.cpp)
list(TRANSFORM HOST_SOURCES PREPEND ${CMAKE_CURRENT_SOURCE_DIR}/)
if(ANDROID)
  # SDL's activity loads libmain.so and calls SDL_main (main.cpp's main, renamed by SDL.h).
  list(APPEND HOST_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/android/android_xbox.cpp)
  add_library(FableXbox SHARED ${HOST_SOURCES})
  set_target_properties(FableXbox PROPERTIES OUTPUT_NAME main)
else()
  add_executable(FableXbox ${HOST_SOURCES})
endif()
target_include_directories(FableXbox PRIVATE ${RUNTIME_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(FableXbox PRIVATE cxx_std_20)
target_compile_options(FableXbox PRIVATE -Wall -Wno-unused-parameter -fno-strict-aliasing)
# glslang compiles the generated NV2A shaders (GLSL) to SPIR-V at run time.
include(FetchContent)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(FABLE_GLSLANG_DIR "" CACHE PATH "Local glslang checkout (default: fetch tag 15.4.0)")
if(FABLE_GLSLANG_DIR)
  add_subdirectory(${FABLE_GLSLANG_DIR} glslang EXCLUDE_FROM_ALL)
else()
  FetchContent_Declare(glslang GIT_REPOSITORY https://github.com/KhronosGroup/glslang.git GIT_TAG 15.4.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(glslang)
endif()

find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
target_include_directories(FableXbox PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../posix/third_party)
target_link_libraries(FableXbox PRIVATE fable_xbox_gen Threads::Threads PkgConfig::SDL2 ${CMAKE_DL_LIBS}
  glslang SPIRV glslang-default-resource-limits)
if(ANDROID)
  set(ADRENOTOOLS_DIR "" CACHE PATH "libadrenotools checkout")
  set(ADRENOTOOLS_BUILD "" CACHE PATH "libadrenotools build directory")
  target_include_directories(FableXbox PRIVATE ${ADRENOTOOLS_DIR}/include)
  target_link_libraries(FableXbox PRIVATE ${ADRENOTOOLS_BUILD}/libadrenotools.a ${ADRENOTOOLS_BUILD}/lib/linkernsbypass/liblinkernsbypass.a log android)
  target_link_options(FableXbox PRIVATE -Wl,-z,max-page-size=16384)
else()
  target_link_options(FableXbox PRIVATE -rdynamic)
endif()
