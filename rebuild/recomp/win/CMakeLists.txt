# Windows x64 build of the recompiled game.
#
# The generated C (from the lifter, run on your own Fable.exe) is not in the
# repository; point FABLE_GEN_DIR at the lifter's output directory.
#
#   cmake -S rebuild/recomp/win -B build/win -G Ninja \
#         -DCMAKE_TOOLCHAIN_FILE=rebuild/recomp/win/cmake/llvm-mingw-x64.cmake -DLLVM_MINGW=<llvm-mingw> \
#         -DFABLE_GEN_DIR=<lifter output> -DCMAKE_BUILD_TYPE=Release
#   cmake --build build/win
cmake_minimum_required(VERSION 3.25)
project(FableRecompWin LANGUAGES C CXX)

set(FABLE_GEN_DIR "" CACHE PATH "Directory with recomp_*.c generated by fable_recomp")
if(NOT EXISTS "${FABLE_GEN_DIR}/recomp_table.c")
  message(FATAL_ERROR "FABLE_GEN_DIR must point at the lifter output (recomp_table.c not found)")
endif()

file(GLOB GEN_SOURCES "${FABLE_GEN_DIR}/recomp_*.c")
set(RUNTIME_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../runtime)

# Generated code: no FP contraction (bit-exact x87 model), aliasing-agnostic memory accessors.
add_library(fable_gen OBJECT ${GEN_SOURCES} ${RUNTIME_DIR}/runtime.c)
target_include_directories(fable_gen PRIVATE ${RUNTIME_DIR} ${FABLE_GEN_DIR})
target_compile_definitions(fable_gen PRIVATE RECOMP_IDENTITY_MEMORY=1)
target_compile_options(fable_gen PRIVATE -O1 -ffp-contract=off -fno-strict-aliasing -w)

# Recompiled helper DLLs from the game folder: "prefix|FileName.dll|<lifter output dir>" each.
set(FABLE_GUEST_DLLS "" CACHE STRING "prefix|file|dir;... for recompiled DLLs")
set(GUEST_DLL_DECLS "")
set(GUEST_DLL_ROWS "")
set(GUEST_DLL_COUNT 0)
foreach(spec ${FABLE_GUEST_DLLS})
  string(REPLACE "|" ";" parts "${spec}")
  list(GET parts 0 pfx)
  list(GET parts 1 dllfile)
  list(GET parts 2 dlldir)
  file(GLOB dll_sources "${dlldir}/${pfx}_*.c")
  target_sources(fable_gen PRIVATE ${dll_sources})
  target_include_directories(fable_gen PRIVATE ${dlldir})
  string(APPEND GUEST_DLL_DECLS "extern const RecompEntry recomp_table_${pfx}[]; extern const uint32_t recomp_table_${pfx}_size;\n")
  string(APPEND GUEST_DLL_ROWS "    {\"${dllfile}\", recomp_table_${pfx}, &recomp_table_${pfx}_size},\n")
  math(EXPR GUEST_DLL_COUNT "${GUEST_DLL_COUNT} + 1")
endforeach()
if(GUEST_DLL_COUNT EQUAL 0)
  set(GUEST_DLL_ROWS "    {0, 0, 0},\n")
endif()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/guest_dlls.c
  "/* Generated by CMake: recompiled DLL tables. */\n#include <stdint.h>\n"
  "typedef void (*GuestFn)(void*);\ntypedef struct RecompEntry { uint32_t addr; GuestFn fn; } RecompEntry;\n"
  "${GUEST_DLL_DECLS}"
  "typedef struct GuestDllTable { const char* file; const RecompEntry* table; const uint32_t* size; } GuestDllTable;\n"
  "const GuestDllTable guest_dll_tables[] = {\n${GUEST_DLL_ROWS}};\nconst uint32_t guest_dll_count = ${GUEST_DLL_COUNT};\n")
target_sources(fable_gen PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/guest_dlls.c)

file(GLOB HOST_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
add_executable(FableRecomp WIN32 ${HOST_SOURCES} $<TARGET_OBJECTS:fable_gen>)
target_include_directories(FableRecomp PRIVATE ${RUNTIME_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(FableRecomp PRIVATE cxx_std_20)
target_compile_options(FableRecomp PRIVATE -ffp-contract=off -fno-strict-aliasing -Wall -Wno-unused-function)
target_link_options(FableRecomp PRIVATE -municode -static -Wl,--high-entropy-va -Wl,--dynamicbase
                    -Wl,--image-base=0x140000000 -Wl,--stack,0x10000000)
target_link_libraries(FableRecomp PRIVATE bcrypt shell32 user32 gdi32 advapi32 winmm imm32 ole32 oleaut32 version ws2_32
                      d3d9 dinput8 dxguid uuid)
